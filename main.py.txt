# -----------------------------
# Generate Legal Document
# -----------------------------
@app.post("/generate", response_class=HTMLResponse)
def generate_document(
    document_type: str,
    name: str,
    second_party: str,
    address: str,
    amount: str,
    duration: str,
    start_date: str
):

    document = f"""
    {document_type.upper()}

    ----------------------------------------

    This document is an AI-assisted draft
    generated by LegalEase.

    FIRST PARTY:
    {name}

    SECOND PARTY:
    {second_party}

    ADDRESS:
    {address}

    AMOUNT:
    {amount}

    DURATION:
    {duration}

    START DATE:
    {start_date}
     ----------------------------------------

    TERMS AND CONDITIONS

    1. Both parties agree to the information
       provided in this document.

    2. Both parties should review the terms
       before signing.

    3. Any applicable legal requirements,
       registration requirements, or local
       rules should be independently verified.

    ----------------------------------------

    AI-GENERATED DRAFT
    For review before execution.

    LegalEase
    """

    # Save generated document
    filename = "generated_document.txt"

    with open(filename, "w", encoding="utf-8") as file:
        file.write(document)

    return f"""
    <!DOCTYPE html>
    <html>

    <head>
        <title>Generated Document</title>

        <style>
            body {{
                font-family: Arial;
                background: #f2f4f7;
                padding: 40px;
            }}

            .document {{
                max-width: 700px;
                margin: auto;
                background: white;
                padding: 40px;
                border-radius: 10px;
            }}

            pre {{
                white-space: pre-wrap;
                font-family: Arial;
                line-height: 1.6;
            }}

            a {{
                display: block;
                text-align: center;
                margin-top: 20px;
                padding: 14px;
                background: #222;
                color: white;
                text-decoration: none;
                border-radius: 6px;
            }}
        </style>

    </head>

    <body>

    <div class="document">

        <h1>Generated Document</h1>

        <pre>{document}</pre>

        <a href="/download">
            Download Document
        </a>

    </div>

    </body>
    </html>
    """


# -----------------------------
# Download Document
# -----------------------------
@app.get("/download")
def download_document():

    return PlainTextResponse(
        open("generated_document.txt", encoding="utf-8").read(),
        headers={
            "Content-Disposition":
            "attachment; filename=LegalEase_Document.txt"
        }
    )
